#+NAME: headers
#+begin_src clojure :results value :var dir=base-directory
  (def files
    (->> (fs/glob "./source" "**{.org}")
         (map str) (map #(subs % (count "source/")))))

  #_(defn mk [value]
      (str "<nav id=\"org-div-home-and-up\">" value "</nav>"))

  (defn mk-a [[link text]]
    (if link
      (format "<a href=\"/%s\">%s</a>" link text)
      (format "<span>%s</span>" text)))

  (defn mk-navbar [as]
    (str
     "<nav id=\"org-div-home-and-up\">"
     (->> as (map mk-a) (clojure.string/join " | "))
     "</nav>"))

  (def main-en ["en.html" "☀️ Main page"])
  (def main-ru ["ru.html" "☀️ Главная страница"])
  (def trans-en "🇺🇸 English version")
  (def trans-ru "🇷🇺 Русская версия")
  (defn org->html [path] (clojure.string/replace path ".org" ".html"))

  (defn header [rel-path]
    (mk-navbar
     (cond
       (clojure.string/starts-with? rel-path "blog/en")
       (let [ruv (clojure.string/replace rel-path "blog/en" "blog/ru")]
         (if (some #{ruv} files)
           [[(clojure.string/replace ruv ".org" ".html") trans-ru] main-en] [main-en]))

       (clojure.string/starts-with? rel-path "blog/ru")
       (let [env (clojure.string/replace rel-path "blog/ru" "blog/en")]
         (if (some #{env} files)
           [[(org->html env) trans-en] main-ru] [main-ru]))

       (= rel-path "index.org")
       [["en.html" trans-en]]

       (= rel-path "ru.org")
       [["en.html" trans-en]]

       (= rel-path "en.org")
       [["ru.html" trans-ru]]

       (= rel-path "about_ru.org")
       [["en.html" trans-en]]

       (= rel-path "about_en.org")
       [["ru.html" trans-ru]]
       )))

  (def table
    (into
     []
     (for [file files]
       [(str dir "source/" file) (header file)]
       )))
  table
#+end_src
